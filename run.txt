# Run from the repository root in PowerShell.
# If ./data/corpus is missing, stage the supplied assessment files:
if (-not (Test-Path .\data\corpus)) {
    New-Item -ItemType Directory -Force .\data | Out-Null
    Copy-Item .\corpus .\data\corpus -Recurse
}
if (-not (Test-Path .\data\manifest.json)) { Copy-Item .\manifest.json .\data\manifest.json }
if (-not (Test-Path .\data\veltrix.db)) { Copy-Item .\veltrix.db .\data\veltrix.db }
if (-not (Test-Path .\.env)) { Copy-Item .\.env.example .\.env }

# Set OPENAI_API_KEY in .env, then build and start the API:
docker compose up --build -d

# Check readiness (HTTP 200 means ingestion is complete):
Invoke-RestMethod http://localhost:8000/health

# Stop the API:
docker compose down
